<script>
(function($) {
  // Configuración dinámica desde WP-Admin
  function getEatGoConfig() {
    var cfg = window.EatGoConfig || {};
    return {
      phone: cfg.whatsappPhone || '50769906424',
      url: cfg.whatsappUrl || 'https://wa.me/50769906424',
      formatted: cfg.whatsappFormatted || '+507 6990-6424',
      deliveryCost: (typeof cfg.deliveryCost !== 'undefined' && cfg.deliveryCost !== null) ? parseFloat(cfg.deliveryCost) : 2.50,
      dispatchHours: cfg.dispatchHours || 'Se despacha de 11:30am a 3:30pm',
      timeRetiro: cfg.timeRetiro || '10-15 minutos',
      timeDelivery: cfg.timeDelivery || '20-35 min'
    };
  }

  // Estado local del Pedido
  var comandaState = {
    deliveryMode: 'delivery', // 'delivery' ($2.50) o 'retiro' ($0.00)
    nombre: '',
    telefono: '',
    direccion: '',
    hora: 'Lo antes posible'
  };

  // Cargar datos previos si existen
  try {
    var savedState = localStorage.getItem('eatgo_order_saved') || localStorage.getItem('eatgo_comanda_saved');
    if (savedState) {
      var parsed = JSON.parse(savedState);
      if (parsed.nombre) comandaState.nombre = parsed.nombre;
      if (parsed.telefono) comandaState.telefono = parsed.telefono;
      if (parsed.direccion) comandaState.direccion = parsed.direccion;
      if (parsed.hora) comandaState.hora = parsed.hora;
      if (parsed.deliveryMode) comandaState.deliveryMode = parsed.deliveryMode;
    }
  } catch(e) {}

  function saveComandaState() {
    try {
      localStorage.setItem('eatgo_order_saved', JSON.stringify({
        deliveryMode: comandaState.deliveryMode,
        nombre: comandaState.nombre,
        telefono: comandaState.telefono,
        direccion: comandaState.direccion,
        hora: comandaState.hora
      }));
    } catch(e) {}
  }

  // Extraer información detallada de los platos en el carrito
  function getCartData() {
    var items = [];
    var subtotalNum = 0;

    $('.elementor-menu-cart__products .elementor-menu-cart__product, .woocommerce-mini-cart .woocommerce-mini-cart-item').each(function() {
      var $item = $(this);
      
      var removeBtn = $item.find('.remove_from_cart_button, .elementor_remove_from_cart_button').first();
      var itemKey = removeBtn.attr('data-cart_item_key') || '';
      var productId = removeBtn.attr('data-product_id') || '';
      var removeUrl = removeBtn.attr('href') || '#';

      var name = $item.find('.elementor-menu-cart__product-name a, .elementor-menu-cart__product-name, .product-name a, .product-name').first().text().trim();
      var imgUrl = $item.find('img').first().attr('src') || '';

      var rawQty = $item.find('.product-quantity, .quantity').text();
      var qtyMatch = rawQty.match(/(\d+)/);
      var qty = qtyMatch ? parseInt(qtyMatch[1], 10) : 1;

      var priceText = $item.find('.amount, .woocommerce-Price-amount').last().text().replace(/[^0-9.]/g, '').trim();
      var unitPrice = parseFloat(priceText) || 0;
      var itemTotal = unitPrice * qty;
      subtotalNum += itemTotal;

      if (name) {
        items.push({
          key: itemKey,
          productId: productId,
          removeUrl: removeUrl,
          name: name,
          imgUrl: imgUrl,
          qty: qty,
          unitPrice: unitPrice,
          total: itemTotal
        });
      }
    });

    var cfg = getEatGoConfig();
    var deliveryCost = (comandaState.deliveryMode === 'delivery') ? cfg.deliveryCost : 0.00;
    var totalNum = (items.length > 0) ? (subtotalNum + deliveryCost) : 0;

    return {
      items: items,
      count: items.reduce(function(acc, it) { return acc + it.qty; }, 0),
      subtotal: subtotalNum,
      deliveryCost: deliveryCost,
      total: totalNum
    };
  }

  // Generar Payload para WhatsApp según especificación exacta
  function buildWhatsAppPayload() {
    var cart = getCartData();
    var cfg = getEatGoConfig();
    var phone = cfg.phone;

    if (cart.items.length === 0) {
      return 'https://api.whatsapp.com/send?phone=' + phone + '&text=' + encodeURIComponent('¡Hola Eat&Go! 🍱 Quiero realizar un pedido.');
    }

    var cliente = comandaState.nombre.trim() || 'Cliente';
    var tel = comandaState.telefono.trim() || 'No indicado';
    var isDelivery = (comandaState.deliveryMode === 'delivery');
    var modalidadText = isDelivery ? 'Delivery - Confirmar área de cobertura' : 'Retiro sin fila - 10-15 min';
    var direccionText = comandaState.direccion.trim() || 'Por coordinar';
    var horaText = comandaState.hora.trim() || 'Lo antes posible';

    var msg = '*RESUMEN DE PEDIDO - EAT & GO* 🍱\n';
    msg += '------------------------------\n';
    msg += '👤 *Cliente:* ' + cliente + '\n';
    msg += '📱 *WhatsApp:* ' + tel + '\n';
    msg += '🛵 *Modalidad:* ' + modalidadText + '\n';
    if (isDelivery) {
      msg += '📍 *Dirección:* ' + direccionText + '\n';
    }
    msg += '⏰ *Hora de retiro:* ' + horaText + ' (Despachos de 11:30am a 3:30pm)\n';
    msg += '------------------------------\n';
    msg += '📋 *DETALLE DEL PEDIDO:*\n';

    cart.items.forEach(function(it) {
      msg += it.qty + 'x ' + it.name + ' ($' + it.total.toFixed(2) + ')\n';
    });

    msg += '------------------------------\n';
    msg += '🍽️ *Cantidad de alimentos:* ' + cart.count + (cart.count === 1 ? ' plato' : ' platos') + '\n';
    msg += '🛵 *Delivery (por confirmar cobertura):* $' + cart.deliveryCost.toFixed(2) + '\n';
    msg += '💰 *TOTAL A PAGAR:* $' + cart.total.toFixed(2) + '\n';
    msg += '------------------------------\n';
    msg += '📲 _Por favor confirmar recepción para enviar a cocina._';

    return 'https://api.whatsapp.com/send?phone=' + phone + '&text=' + encodeURIComponent(msg);
  }

  // Sincronizar enlaces globales de WhatsApp con la configuración de WP-Admin
  function syncGlobalWhatsAppLinks() {
    var cfg = getEatGoConfig();
    if (!cfg.phone) return;
    $('a[href*="wa.me"]').each(function() {
      var $a = $(this);
      if (!$a.hasClass('eg-btn-whatsapp')) {
        $a.attr('href', cfg.url);
        var text = $a.text().trim();
        if (text.indexOf('507') !== -1 || text.indexOf('6990') !== -1 || text.indexOf('6234') !== -1) {
          $a.text(cfg.formatted);
        }
      }
    });
  }

  // Renderizar / Sincronizar UI del Resumen de Pedido (Side-Drawer)
  function renderComandaDigital() {
    renderEatGoFabCart();
    syncGlobalWhatsAppLinks();
    var $main = $('.elementor-menu-cart__main');
    if (!$main.length) return;

    var cart = getCartData();
    var cfg = getEatGoConfig();

    // 1. Inyectar Cabecera Carbón si no existe
    var $header = $main.find('.eg-comanda-header');
    if (!$header.length) {
      var headerHtml = '<div class="eg-comanda-header">' +
        '<div class="eg-header-left">' +
          '<div class="eg-header-badge">EG</div>' +
          '<div class="eg-header-titles">' +
            '<h3 class="eg-header-title">Resumen de pedido</h3>' +
            '<span class="eg-header-count"></span>' +
          '</div>' +
        '</div>' +
        '<div class="eg-header-right">' +
          '<button type="button" class="eg-btn-vaciar">Vaciar</button>' +
          '<button type="button" class="eg-btn-close">✕</button>' +
        '</div>' +
      '</div>';
      $main.prepend(headerHtml);
      $header = $main.find('.eg-comanda-header');
    }

    // Actualizar contador en cabecera
    var countLabel = cart.count === 1 ? '1 plato en tu pedido' : (cart.count + ' platos en tu pedido');
    $header.find('.eg-header-count').text(cart.count > 0 ? countLabel : 'Tu pedido está vacío');
    if (cart.count === 0) {
      $header.find('.eg-btn-vaciar').hide();
    } else {
      $header.find('.eg-btn-vaciar').show();
    }

    var $content = $main.find('.widget_shopping_cart_content');
    if (!$content.length) return;

    // Si el carrito está vacío:
    if (cart.items.length === 0) {
      $content.find('.eg-comanda-body').remove();
      if (!$content.find('.eg-empty-cart').length) {
        var emptyHtml = '<div class="eg-empty-cart">' +
          '<div class="eg-empty-icon">🍱</div>' +
          '<div class="eg-empty-title">Tu comanda está vacía</div>' +
          '<p class="eg-empty-desc">Agrega tus platos favoritos desde el Menú Semanal o Menú del Día para despachar a tu oficina.</p>' +
          '<a href="/menu-semanal/" class="eg-empty-btn">Ver Menú Semanal →</a>' +
        '</div>';
        $content.append(emptyHtml);
      }
      return;
    }

    // Si hay platos: remover estado vacío
    $content.find('.eg-empty-cart').remove();

    var $productContainer = $content.find('.elementor-menu-cart__products, .woocommerce-mini-cart');
    
    // Inyectar título de sección encima de la lista de platos
    if (!$productContainer.prev('.eg-section-header-platos').length) {
      $productContainer.before('<div class="eg-section-header eg-section-header-platos">PLATOS EN LA COMANDA</div>');
    }

    // Iterar cada tarjeta de producto de Elementor
    $productContainer.find('.elementor-menu-cart__product, .woocommerce-mini-cart-item').each(function() {
      var $item = $(this);
      
      if ($item.find('.eg-custom-card').length) {
        var existingKey = $item.find('.eg-qty-selector').attr('data-key');
        var matchingItem = cart.items.find(function(it) { return it.key === existingKey; });
        if (matchingItem) {
          $item.find('.eg-qty-val').text(matchingItem.qty);
          $item.find('.eg-card-price').text('$' + matchingItem.total.toFixed(2));
        }
        return;
      }

      var name = $item.find('.elementor-menu-cart__product-name a, .elementor-menu-cart__product-name, .product-name a, .product-name').first().text().trim();
      var imgUrl = $item.find('img').first().attr('src') || '';
      var removeBtn = $item.find('.remove_from_cart_button, .elementor_remove_from_cart_button').first();
      var itemKey = removeBtn.attr('data-cart_item_key') || '';
      var rawQty = $item.find('.product-quantity, .quantity').text();
      var qtyMatch = rawQty.match(/(\d+)/);
      var qty = qtyMatch ? parseInt(qtyMatch[1], 10) : 1;
      var priceText = $item.find('.amount, .woocommerce-Price-amount').last().text().replace(/[^0-9.]/g, '').trim();
      var unitPrice = parseFloat(priceText) || 0;
      var totalItem = unitPrice * qty;

      var cardHtml = '<div class="eg-custom-card">' +
        '<div class="eg-card-top">' +
          '<div class="eg-card-info">' +
            (imgUrl ? '<img src="' + imgUrl + '" class="eg-card-thumb" alt="' + name + '" />' : '') +
            '<div class="eg-card-texts">' +
              '<h4 class="eg-card-title">' + name + '</h4>' +
              '<span class="eg-card-subtext">Empaque térmico sellado incluido</span>' +
            '</div>' +
          '</div>' +
          '<div class="eg-card-price">$' + totalItem.toFixed(2) + '</div>' +
        '</div>' +
        '<div class="eg-card-bottom">' +
          '<div class="eg-qty-selector" data-key="' + itemKey + '" data-qty="' + qty + '">' +
            '<button type="button" class="eg-qty-btn eg-qty-minus">−</button>' +
            '<span class="eg-qty-val">' + qty + '</span>' +
            '<button type="button" class="eg-qty-btn eg-qty-plus">+</button>' +
          '</div>' +
          '<button type="button" class="eg-item-delete" data-key="' + itemKey + '" title="Eliminar plato">' +
            '<svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><polyline points="3 6 5 6 21 6"></polyline><path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2"></path></svg>' +
          '</button>' +
        '</div>' +
      '</div>';

      $item.append(cardHtml);
    });

    // 3. Inyectar Bloque Interactivo (Modalidad, Datos de Pedido, Resumen y Acción Principal)
    var $comandaBody = $content.find('.eg-comanda-body');
    if (!$comandaBody.length) {
      var bodyHtml = '<div class="eg-comanda-body">' +
        // Modalidad de entrega
        '<div class="eg-section-header">MODALIDAD DE ENTREGA</div>' +
        '<div class="eg-modalidad-grid">' +
          '<div class="eg-modalidad-card eg-card-retiro" data-mode="retiro">' +
            '<div class="eg-modalidad-title">🏪 Retiro sin fila</div>' +
            '<div class="eg-modalidad-sub">' + cfg.timeRetiro + ' ($0.00)</div>' +
          '</div>' +
          '<div class="eg-modalidad-card eg-card-delivery is-active" data-mode="delivery">' +
            '<div class="eg-modalidad-title">🛵 Delivery</div>' +
            '<div class="eg-modalidad-sub">Confirmar cobertura • ' + cfg.timeDelivery + ' ($' + cfg.deliveryCost.toFixed(2) + ')</div>' +
          '</div>' +
        '</div>' +

        // Formulario de datos del pedido
        '<div class="eg-section-header">DATOS DEL PEDIDO</div>' +
        '<div class="eg-despacho-form">' +
          '<div class="eg-form-row">' +
            '<input type="text" class="eg-form-input eg-inp-nombre" placeholder="Nombre" value="' + (comandaState.nombre || '') + '">' +
            '<input type="tel" class="eg-form-input eg-inp-tel" placeholder="Teléfono" value="' + (comandaState.telefono || '') + '">' +
          '</div>' +
          '<div class="eg-form-row eg-row-direccion" style="grid-column: 1 / -1;">' +
            '<input type="text" class="eg-form-input eg-inp-direccion" placeholder="Dirección completa" value="' + (comandaState.direccion || '') + '">' +
          '</div>' +
          '<div class="eg-form-row" style="grid-column: 1 / -1;">' +
            '<div style="width:100%;">' +
              '<input type="text" class="eg-form-input eg-inp-hora" list="eg-horas-list" placeholder="Hora de retiro (ej: 12:30 PM)" value="' + (comandaState.hora || '') + '">' +
              '<datalist id="eg-horas-list">' +
                '<option value="Lo antes posible"></option>' +
                '<option value="11:30 AM"></option>' +
                '<option value="12:00 MD"></option>' +
                '<option value="12:30 PM"></option>' +
                '<option value="1:00 PM"></option>' +
                '<option value="1:30 PM"></option>' +
                '<option value="2:00 PM"></option>' +
                '<option value="2:30 PM"></option>' +
                '<option value="3:00 PM"></option>' +
                '<option value="3:30 PM"></option>' +
              '</datalist>' +
              '<div class="eg-help-hours" style="color: #6B7280; font-size: 11px; margin-top: 4px; line-height: 1.3;">' + cfg.dispatchHours + '</div>' +
            '</div>' +
          '</div>' +
        '</div>' +

        // Resumen de pago
        '<div class="eg-section-header">RESUMEN DE PAGO</div>' +
        '<div class="eg-resumen-box">' +
          '<div class="eg-resumen-row">' +
            '<span>Cantidad de alimentos:</span>' +
            '<strong class="eg-val-alimentos">' + cart.count + (cart.count === 1 ? ' plato' : ' platos') + '</strong>' +
          '</div>' +
          '<div class="eg-resumen-row eg-row-delivery-cost">' +
            '<span>Delivery (por confirmar cobertura):</span>' +
            '<strong class="eg-val-delivery">$' + cart.deliveryCost.toFixed(2) + '</strong>' +
          '</div>' +
          '<div class="eg-resumen-row is-divider"></div>' +
          '<div class="eg-resumen-row is-total">' +
            '<span class="eg-resumen-total-label">TOTAL A PAGAR:</span>' +
            '<span class="eg-resumen-total-val">$' + cart.total.toFixed(2) + '</span>' +
          '</div>' +
        '</div>' +

        // Botón de acción principal limpio
        '<div class="eg-comanda-actions">' +
          '<a href="' + buildWhatsAppPayload() + '" target="_blank" class="eg-btn-whatsapp">' +
            '<svg width="22" height="22" viewBox="0 0 24 24" fill="currentColor" style="display:inline-block; vertical-align:middle; margin-right:4px;"><path d="M12.031 6.172c-3.181 0-5.767 2.586-5.768 5.766-.001 1.298.38 2.27 1.019 3.287l-.582 2.128 2.182-.573c.978.58 1.911.928 3.145.929 3.178 0 5.767-2.587 5.768-5.766.001-3.187-2.575-5.77-5.764-5.771zm3.392 8.244c-.144.405-.837.774-1.17.824-.312.045-.698.074-2.146-.525-1.745-.722-2.883-2.484-2.97-2.6-.088-.115-.705-.939-.705-1.792 0-.853.447-1.272.607-1.446.16-.174.349-.218.465-.218.117 0 .233.001.334.006.107.005.25.04.39.377.145.349.494 1.206.538 1.294.043.088.073.19.014.307-.058.117-.087.19-.174.292-.087.102-.184.228-.263.307-.087.087-.179.182-.077.357.102.175.454.749.974 1.213.668.595 1.232.78 1.406.867.175.088.277.073.379-.044.102-.116.438-.51.554-.685.117-.175.234-.146.393-.087.16.058 1.018.48 1.193.568.175.087.291.131.334.204.044.073.044.423-.1 1.028z"/></svg>' +
            'Realizar pedido' +
          '</a>' +
        '</div>' +
      '</div>';

      $content.append(bodyHtml);
      $comandaBody = $content.find('.eg-comanda-body');
    }

    // 4. Actualizar dinámicamente los valores del Bloque Interactivo
    $comandaBody.find('.eg-val-alimentos').text(cart.count + (cart.count === 1 ? ' plato' : ' platos'));
    $comandaBody.find('.eg-val-delivery').text('$' + cart.deliveryCost.toFixed(2));
    $comandaBody.find('.eg-resumen-total-val').text('$' + cart.total.toFixed(2));

    // Estado visual y condicional de modalidad
    $comandaBody.find('.eg-modalidad-card').removeClass('is-active');
    $comandaBody.find('.eg-modalidad-card[data-mode="' + comandaState.deliveryMode + '"]').addClass('is-active');

    if (comandaState.deliveryMode === 'retiro') {
      $comandaBody.find('.eg-row-direccion').addClass('is-hidden');
      $comandaBody.find('.eg-val-delivery').text('$0.00');

      $comandaBody.find('.eg-row-direccion').removeClass('is-hidden');
      $comandaBody.find('.eg-val-delivery').text('$' + cfg.deliveryCost.toFixed(2));
    }

  function renderEatGoFabCart() {
    var $fab = $('.eatgo-fab-cart');
    if (!$fab.length) {
      var fabHtml = [
        '<a href="#comanda" class="eatgo-fab-cart" role="button" aria-label="Ver Resumen de Pedido">',
          '<span class="eatgo-fab-icon">',
            '<svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="#FFFFFF" stroke-width="2.3" stroke-linecap="round" stroke-linejoin="round">',
              '<path d="M6 2L3 6v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2V6l-3-4z"></path>',
              '<line x1="3" y1="6" x2="21" y2="6"></line>',
              '<path d="M16 10a4 4 0 0 1-8 0"></path>',
            '</svg>',
          '</span>',
          '<span class="eatgo-fab-badge">0</span>',
          '<span class="eatgo-fab-total">$0.00</span>',
        '</a>'
      ].join('');
      $('body').append(fabHtml);
      $fab = $('.eatgo-fab-cart');
    }

    var cart = (typeof getCartData === 'function') ? getCartData() : { count: 0, subtotal: 0 };
    var count = cart.count;
    var totalStr = '$' + (cart.subtotal || 0).toFixed(2);

    if (count === 0) {
      var elemQty = parseInt($('.elementor-button-icon-qty').attr('data-counter') || $('.elementor-button-icon-qty').text() || '0', 10);
      var elemAmount = $('.elementor-button-text .woocommerce-Price-amount').first().text().trim();
      if (elemQty > 0) {
        count = elemQty;
        totalStr = elemAmount || '$0.00';
      }
    }

    $fab.find('.eatgo-fab-badge').text(count);
    $fab.find('.eatgo-fab-total').text(totalStr);

    if (count > 0) {
      $fab.css('display', 'inline-flex');
    } else {
      $fab.hide();
    }
  }

  function openSideComandaDrawer() {
    if ($('.woofc-cart-toggle').length) {
      $('.woofc-cart-toggle').trigger('click');
    } else if (typeof xoo_wsc_cart !== 'undefined') {
      xoo_wsc_cart.open();
    } else if ($('.elementor-menu-cart__toggle, #elementor-menu-cart__toggle_button').length) {
      $('.elementor-menu-cart__toggle, #elementor-menu-cart__toggle_button').first().trigger('click');
      $('.elementor-widget-woocommerce-menu-cart, .elementor-menu-cart__container').addClass('elementor-menu-cart--shown').attr('aria-hidden', 'false');
      if (typeof renderComandaDigital === 'function') {
        renderComandaDigital();
      }
    } else if (window.elementorProFrontend && elementorProFrontend.modules && elementorProFrontend.modules.popup) {
      elementorProFrontend.modules.popup.showPopup({ id: 326 });
    }
  }

  // Click en FAB para abrir Side Drawer
  $(document).on('click', '.eatgo-fab-cart, .eatgo-fab-cart *', function(e) {
    e.preventDefault();
    openSideComandaDrawer();

  // Vinculación modal de producto (WPC Quick View)
  function syncCardQuickViewButtons() {
    $('.e-loop-item.product, .e-loop-item[class*="post-"], .e-loop-item[class*="e-loop-item-"]').each(function() {
      var $card = $(this);
      var match = ($card.attr('class') || '').match(/\bpost-(\d+)\b/);
      if (!match) {
        match = ($card.attr('class') || '').match(/\be-loop-item-(\d+)\b/);
      }
      if (match && match[1]) {
        var pid = match[1];
        var $btn = $card.find('.elementor-element-w14btna a, .elementor-element-9d0e1f8 a, a.woosq-btn, a.elementor-button:has(.elementor-button-text:contains("Agregar"))');
        $btn.addClass('woosq-btn').attr('data-id', pid).attr('href', '#');
      }
    });
  }

  $(document).on('click', '.woosq-btn, .woosq-btn *, .elementor-element-w14btna a, .elementor-element-w14btna a *, .elementor-element-9d0e1f8 a, .elementor-element-9d0e1f8 a *, a.elementor-button:has(.elementor-button-text:contains("Agregar")), .elementor-button-text:contains("Agregar")', function(e) {
    e.preventDefault();
    var $el = $(this);
    var $btn = $el.closest('a');
    var pid = $btn.attr('data-id');
    if (!pid || isNaN(pid)) {
      var $card = $el.closest('.e-loop-item, [class*="post-"]');
      var match = ($card.attr('class') || '').match(/(?:post-|e-loop-item-)(\d+)/);
      if (match && match[1]) pid = match[1];
    }
    if (pid && !isNaN(pid)) {
      $btn.attr('data-id', pid).addClass('woosq-btn').attr('href', '#');
      if (typeof woosq_open === 'function') {
        woosq_open(pid);
      } else {
        $(document.body).trigger('woosq_open', [pid]);
      }
    }
  });

  // EVENTOS INTERACTIVOS DEL CAJÓN
  $(document).on('click', '.eg-btn-close', function(e) {
    e.preventDefault();
    $('.elementor-menu-cart__close-button').trigger('click');
    $('.elementor-widget-woocommerce-menu-cart, .elementor-menu-cart__container').removeClass('elementor-menu-cart--shown').attr('aria-hidden', 'true');
  });

  $(document).on('click', '.eg-btn-vaciar', function(e) {
    e.preventDefault();
    if (!confirm('¿Deseas vaciar todos los platos de tu pedido?')) return;
    
    var $btn = $(this);
    $btn.text('Vaciando...').css('opacity', '0.5');

    $.ajax({
      url: '/wp-admin/admin-ajax.php',
      type: 'POST',
      data: { action: 'eatgo_cart_clear' },
      success: function(resp) {
        $(document.body).trigger('wc_fragment_refresh');
        renderComandaDigital();
      },
      complete: function() {
        $btn.text('Vaciar').css('opacity', '1');
      }
    });
  });

  $(document).on('click', '.eg-qty-btn', function(e) {
    e.preventDefault();
    var $btn = $(this);
    var $selector = $btn.closest('.eg-qty-selector');
    var itemKey = $selector.attr('data-key');
    var currentQty = parseInt($selector.attr('data-qty'), 10) || 1;
    var isPlus = $btn.hasClass('eg-qty-plus');
    var newQty = isPlus ? (currentQty + 1) : (currentQty - 1);

    $selector.css('opacity', '0.5');

    $.ajax({
      url: '/wp-admin/admin-ajax.php',
      type: 'POST',
      data: {
        action: 'eatgo_cart_update_qty',
        cart_item_key: itemKey,
        quantity: newQty
      },
      success: function() {
        $(document.body).trigger('wc_fragment_refresh');
      }
    });
  });

  $(document).on('click', '.eg-item-delete', function(e) {
    e.preventDefault();
    var itemKey = $(this).attr('data-key');
    var $card = $(this).closest('.elementor-menu-cart__product, .woocommerce-mini-cart-item');
    $card.css('opacity', '0.4');

    $.ajax({
      url: '/wp-admin/admin-ajax.php',
      type: 'POST',
      data: {
        action: 'eatgo_cart_update_qty',
        cart_item_key: itemKey,
        quantity: 0
      },
      success: function() {
        $(document.body).trigger('wc_fragment_refresh');
      }
    });
  });

  // Conmutar modalidad de entrega
  $(document).on('click', '.eg-modalidad-card', function() {
    var mode = $(this).attr('data-mode');
    comandaState.deliveryMode = mode;
    saveComandaState();
    renderComandaDigital();
    renderEatGoFabCart();
    syncCardQuickViewButtons();
  });

    saveComandaState();

  $(document).on('input change', '.eg-inp-direccion', function() {
    comandaState.direccion = $(this).val();
    saveComandaState();
    $('.eg-btn-whatsapp').attr('href', buildWhatsAppPayload());
  });
  $(document).on('input change', '.eg-inp-hora', function() {
    comandaState.hora = $(this).val();
    saveComandaState();
    $('.eg-btn-whatsapp').attr('href', buildWhatsAppPayload());
  });

  // Botón WhatsApp: regenerar link al hacer clic para asegurar datos más frescos
  $(document).on('click', '.eg-btn-whatsapp', function() {
    $(this).attr('href', buildWhatsAppPayload());
  });

  // Inicialización y reactividad en eventos de WooCommerce y Elementor
  $(document).ready(function() {
    renderEatGoFabCart();
    renderComandaDigital();
    syncCardQuickViewButtons();
    syncGlobalWhatsAppLinks();

  $(document).on('added_to_cart removed_from_cart wc_fragments_refreshed wc_fragments_loaded elementor/popup/show', function() {
    setTimeout(function() { 
      renderComandaDigital(); 
      renderEatGoFabCart(); 
      syncGlobalWhatsAppLinks();
    }, 40);
  });

  renderEatGoFabCart();
  syncCardQuickViewButtons();
  syncGlobalWhatsAppLinks();

})(jQuery);
